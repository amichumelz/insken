        const types = ['Physical', 'Online'];
        
        let count = 1248;

        function createFeedItem() {
            const name = names[Math.floor(Math.random() * names.length)];
            const sector = sectors[Math.floor(Math.random() * sectors.length)];
            const type = types[Math.floor(Math.random() * types.length)];
            const isPhysical = type === 'Physical';
            
            const time = new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit', second:'2-digit'});
            
            const div = document.createElement('div');
            div.className = 'flex items-center justify-between p-3 bg-surface-container-lowest hover:bg-surface-container-low transition-colors border-b border-surface-container rounded fade-in-up';
            
            div.innerHTML = `
                <div class="flex items-center gap-3">
                    <div class="w-10 h-10 rounded-full bg-${isPhysical ? 'primary' : 'secondary'}-container flex items-center justify-center text-${isPhysical ? 'primary' : 'secondary'}">
                        <span class="material-symbols-outlined text-sm" data-icon="${isPhysical ? 'meeting_room' : 'devices'}">${isPhysical ? 'meeting_room' : 'devices'}</span>
                    </div>
                    <div>
                        <div class="font-body-sm font-semibold text-on-surface">${name}</div>
                        <div class="font-body-sm text-xs text-on-surface-variant">${sector} • ASN-24-${Math.floor(1000 + Math.random() * 9000)}</div>
                    </div>
                </div>
                <div class="text-right">
                    <div class="font-label-caps text-[10px] uppercase ${isPhysical ? 'text-primary bg-primary-fixed' : 'text-secondary bg-secondary-fixed'} px-2 py-0.5 rounded inline-block mb-1">
                        Attended_${type}
                    </div>
                    <div class="font-data-mono text-xs text-outline-variant block">${time}</div>
                </div>
            `;
            
            return div;
        }

        // Initialize feed with some items
        for(let i=0; i<6; i++) {
            const item = createFeedItem();
            item.style.animation = 'none'; // remove animation for initial load
            item.style.opacity = '1';
            feedContainer.appendChild(item);
        }

        // Simulate live updates
        setInterval(() => {
            const newItem = createFeedItem();
            feedContainer.insertBefore(newItem, feedContainer.firstChild);
            
            // Keep max items to prevent infinite scroll
            if(feedContainer.children.length > 20) {
                feedContainer.removeChild(feedContainer.lastChild);
            }
            
            // Update counter occasionally
            if(Math.random() > 0.5) {
                count++;
                counterEl.innerText = count.toLocaleString();
            }
            
        }, 3500);
        
        // CSS specific to tailwind arbitrary values added via JS
        tailwind.config.theme.extend.keyframes = {
            shimmer: {
                '100%': { transform: 'translateX(100%)' }
            }
        };
    </script>
